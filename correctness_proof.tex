\begin{frame}{Functional Correctness Proof}
  \begin{itemize}
      \setlength{\itemsep}{4mm}
      \item \textbf{Goal}: Prove that the Jasmin implementation of \xmssmt adheres to the
    RFC-like \EasyCrypt specification.
    \item Extract an \EasyCrypt representation of the Jasmin program
      using the Jasmin compiler. Program equivalence is then expressed via \textbf{pRHL} statements.
    \item We assume the functional correctness of the underlying SHA-256 implementation.
  \end{itemize}
\end{frame}

\begin{frame}[fragile]
  {Functional Correctness: Relating Representations}
  \begin{itemize}
      \setlength{\itemsep}{4mm}
    \item \textbf{Specification:} records for keys and signatures.
    \item \textbf{Implementation:} byte arrays and raw pointers.
    \item Encoding and decoding operators connect these representations
  \end{itemize}

\begin{frame}[fragile]
  {Functional Correctness Proof: Reasoning About Memory}
  \begin{itemize}
    \item Raw pointers require explicit reasoning about address arithmetic
      and the contents of memory (e.g. pointers don't overlap and do not overflow the 64-bit address space).
  \end{itemize}
  \vspace{2mm}
    \begin{figure}[t!]
    \centering
    \begin{minipage}{.6\textwidth}
      \begin{easycrypt}[basicstyle=\fontsize{7}{8}\selectfont\ttfamily]{}{}
pred valid_ptr_i (p : W64.t) (o : int) = 
  0 <= o => 
    0 <= to_uint p /\ to_uint (p) + o < W64.modulus.

pred disjoint_ptr (p1 l1 p2 l2 : int) =
  forall (k1 : int), 0 <= k1 < l1 =>
    forall (k2 : int), 0 <= k2 < l2 =>
      p1 + k1 <> p2 + k2.\end{easycrypt}
    \end{minipage}
    \caption{Pointer predicates in \EasyCrypt.}
  \end{figure}
\end{frame}
