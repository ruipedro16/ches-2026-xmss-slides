\begin{frame}{Implementation: Overview}
  \begin{itemize}
      \setlength{\itemsep}{2mm}
    \item \xmssmt implementation written in Jasmin targeting x86\_64, based on
      the C reference
      implementation\footnote[frame]{\tiny\url{https://github.com/XMSS/xmss-reference}},
      prioritizing readability and closeness to RFC~8391.
    \item \textbf{SHA-256 from
      libjade}\footnote{\tiny\url{https://github.com/formosa-crypto/libjade}}
      (the C reference uses OpenSSL).
    \item \textbf{Fixed-size data:} stack arrays with sizes known at compile
      time; a separate hash function for each input length.
    \item \textbf{Variable-size data:} raw pointers with explicit
      loads and stores, e.g. for the message to be signed.
    \item \textbf{Arrays:} array arguments must be disjoint.
  \end{itemize}
\end{frame}
\endgroup

\begin{frame}{Implementation: Authentication Paths}
  \begin{itemize}
      \setlength{\itemsep}{4mm}
    \item \textbf{TreeHash algorithm:} Used to compute the root of a
      subtree: keep intermediate nodes on a stack;
      merge the top two nodes whenever they have the same height.
    \item \textbf{Jasmin impl. follows RFC~8391:} compute each node
      of the authentication-path
      with a separate TreeHash call on the corresponding subtree.
    \item \textbf{C reference:} compute the root and authentication path
      together in a single TreeHash traversal.
    \item \textbf{Tradeoff}: Closer correspondence to the RFC
      simplifies the correctness proof, but increases execution time.
  \end{itemize}
\end{frame}

\begin{frame}{Implementation: Constant-Time and Zeroization}
  \begin{itemize}
      \setlength{\itemsep}{4mm}
    \item \textbf{Constant-time validated with Jasmin's CT checker.}
      Annotate public values loaded from memory (e.g. message length):
      memory loads are secret by default, and need to be explicitly
      marked as public.
    \item \textbf{Stack zeroization on return from exported functions:}
      compiler-generated code clears the stack, ensuring that no
      secret data remains on the stack after
      function return.
      \begin{itemize}
          \setlength{\itemsep}{1mm}
        \item Loop zeroization adds less than $0.4\%$ overhead for
          all three operations.
          Unrolling adds about 30\,KB of code and is slightly slower in these
          benchmarks.
      \end{itemize}
  \end{itemize}
\end{frame}
