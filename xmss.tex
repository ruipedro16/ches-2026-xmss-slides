\begin{frame}{\xmss: Overview}
  \begin{itemize}
      \setlength{\itemsep}{4mm}
    \item XMSS~\cite{DBLP:conf/pqcrypto/BuchmannDH11} is a stateful
      hash-based signature scheme.
    \item Security relies on the security of the underlying hash function.
    \item XMSS combines a one-time signature scheme (\wots) with a Merkle tree.
      \vspace{2mm}
      \begin{itemize}
          \setlength{\itemsep}{1mm}
        \item The root of the Merkle tree is the public key of the
          XMSS instance.
        \item Each leaf corresponds to a (compressed) \wots public key.
        \item A Merkle tree of height $h$ has $2^h$ leaves, and thus
          allows signing $2^h$ messages.
      \end{itemize}
  \end{itemize}
\end{frame}

% \wots key generation and signing: RFC 8391, Sections 3.1.3--3.1.5.
\begin{frame}{\wots: Key Generation}
  \footnotesize
  \begin{enumerate}
      \setlength{\itemsep}{1mm}
    \item Generate $\ell$ secret $n$-byte values
      $\mathsf{sk}_0,\ldots,\mathsf{sk}_{\ell-1}$ (pseudorandomly)
    \item Each value starts a hash chain: apply the chaning function
      $w-1$ steps to obtain
      % $\mathsf{pk}_j=\mathsf{chain}(\mathsf{sk}_j,0,w-1)$.
    \item The chaining function is defined as $c_k^i(x,\mathbf{r}) =
      f_k(c_k^{i-1}(x,\mathbf{r})\oplus{r_i})$, with $c_k^0(x,\mathbf{r}) = x$,
      where $\mathbf{r}$ are randomizers and $k$ is keying parameter
      for the hash function.\\
  \end{enumerate}
  \vspace{0mm}
  \begin{center}
    \centering
    \begin{tikzpicture}[x=1.05cm,y=0.65cm,font=\scriptsize,>=Latex]
      \foreach \j/\y in {0/1,1/0,\ell-1/-1} {
        \node[anchor=east,text=Gold] at (-0.3,\y) {$\mathsf{sk}_{\j}$};
        \foreach \k in {0,...,3}
        \node[xmss node] (k\y\k) at (\k,\y) {};
        \foreach \k/\next in {0/1,1/2,2/3}
        \draw[->,Teal] (k\y\k) -- (k\y\next);
        \node[anchor=west,text=Teal] at (3.3,\y) {$\mathsf{pk}_{\j}$};
      }
      \node at (1.5,-1.65) {Example: $w=4$};
    \end{tikzpicture}
  \end{center}
  \vspace{1mm}
  {\footnotesize
    \textbf{Secret key:}
    $\mathsf{SK}=(\mathsf{sk}_0,\ldots,\mathsf{sk}_{\ell-1})$\\[1mm]
    \textbf{Public key:}
    $\mathsf{PK}=(\mathsf{pk}_0,\ldots,\mathsf{pk}_{\ell-1})$\\[1mm]
    $n$: element size in bytes; $w$: Winternitz parameter;
    $\ell$: number of message and checksum digits.\\[1mm]
  }
\end{frame}

\begin{frame}{\wots: Signature Generation \& Verification}
  \small
  \begin{enumerate}
      \setlength{\itemsep}{2mm}
    \item Encode the message as base-$w$ digits
      $m_0,\ldots,m_{\ell_1-1}$
    \item Append the base-$w$ digits of the checksum
      $C=\sum_{j=0}^{\ell_1-1}(w-1-m_j)$, giving
      $\ell$ digits $b_0,\ldots,b_{\ell-1}\in\{0,\ldots,w-1\}$.
    \item The signature is $(\sigma_0,\ldots,\sigma_{\ell-1})$, where
      $\sigma_j=c_k^{b_j}(\mathsf{sk}_j,\mathbf{r})$.
    \item To verify a signature, it suffices to
      confirm that completing the chains permits reconstructing the
      \wots public key.
  \end{enumerate}
  \vspace{2mm}
  \begin{center}
    \begin{tikzpicture}[x=1.7cm,y=0.75cm,font=\scriptsize,>=Latex]
      \foreach \j/\y/\digit in {0/1/1,1/0/3,2/-1/2} {
        \node[anchor=east] at (-0.3,\y) {$b_{\j}=\digit$};
        \foreach \k in {0,...,3}
        \node[xmss node,draw=Muted,fill=white] (s\j\k) at (\k,\y) {};
        \foreach \k/\next in {0/1,1/2,2/3} {
          \ifnum\k<\digit
          \draw[->,Teal,thick] (s\j\k) -- (s\j\next);
          \else
          \draw[->,Muted,dashed] (s\j\k) -- (s\j\next);
          \fi
        }
        \node[xmss node,draw=Gold,fill=Gold!20] at (s\j\digit) {};
        \node[anchor=west,text=Gold] at (3.3,\y) {$\sigma_{\j}$};
      }
      \node at (1.5,-1.65) {Example: $w=4$; gold nodes form the signature.};
    \end{tikzpicture}
  \end{center}
  {\footnotesize
    \alert{Each \wots keypair signs at most one message.}
  }
\end{frame}

% L-tree compression: RFC 8391, Section 4.1.5, Algorithm 8.
\begin{frame}{\ltree: Compressing a \wots Public Key}
  \small
  \begin{columns}[T,onlytextwidth]
    \begin{column}{0.55\textwidth}
      \textbf{Input:} $\ell$ public-key elements, each $n$ bytes.
      \begin{enumerate}
          \setlength{\itemsep}{2.5mm}
        \item Hash adjacent pairs with randomized hashing (hash keys
            and masks are derived from
          \texttt{pub\_seed} and the \ltree address).
        \item If a level has an odd number of nodes, carry the
          last node to the next level unchanged.
        \item Repeat until a single $n$-byte node remains.
      \end{enumerate}

    \end{column}

\begin{frame}{\xmss: Key Generation}
  \footnotesize
  \begin{enumerate}
      \setlength{\itemsep}{0.5mm}
    \item Sample $n$-byte seeds \texttt{sk\_seed}, \texttt{sk\_prf},
      and \texttt{pub\_seed}; initialize the signing index $i=0$.
    \item Derive \wots secret keys via a PRF keyed by
      \texttt{sk\_seed}; compute and compress (via \ltree) the $2^h$
      public keys.
    \item Build the \xmss tree by hashing pairs of sibling nodes
      together to obtain the root.
      A PRF keyed by \texttt{pub\_seed} derives hash keys and
      bitmasks using the node address.
  \end{enumerate}
  \vspace{1mm}
  {\centering \resizebox{0.48\textwidth}{!}{\xmsskeygentree}\par}
  \vspace{2mm}
  {\footnotesize
    \textbf{Secret key:}
    $\mathsf{SK}=(i,\mathtt{sk\_seed},\mathtt{sk\_prf},\mathsf{PK})$
    \hfill
    \textbf{Public key:} $\mathsf{PK}=(\mathsf{root},\mathtt{pub\_seed})$\\[1mm]
    \texttt{sk\_prf}: PRF key for message-hashing randomizers during signing.
  }
\end{frame}

\begin{frame}{\xmss: Signature Generation}
  \small
  \begin{enumerate}
      \setlength{\itemsep}{2mm}
    \item Set the index $i$ to the current secret key index:
      (indicates the one-time \wots key used).
    \item Update the index in the secret key to prevent key reuse.
    \item Derive a pseudorandom value $R$ using a PRF keyed by
      \texttt{sk\_prf}, then compute the message digest using $R$.
    \item Sign the digest with the selected \wots secret key.
    \item Compute the authentication path for leaf $i$:
      it authenticates the corresponding \wots public key in the \xmss tree.
    \item The signature is
      $\sigma=(i,R,\sigma_{\mathrm{WOTS}^{+}},\mathsf{auth})$
  \end{enumerate}
\end{frame}

\begin{frame}{\xmss: Signature Verification}
  \relax
  {\small
    \textbf{Inputs}: message $M$, $\mathsf{PK}=(\mathsf{root},\mathsf{SEED})$,
    $\sigma=(i,R,\sigma_{\mathrm{\wots}},\mathsf{auth}_i)$, with $0\leq
  i<2^h$.\par}
  \vspace{2mm}
  \begin{columns}[T,onlytextwidth]
    \begin{column}{0.54\textwidth}
      \begin{enumerate}
          \small
          \setlength{\itemsep}{0.5mm}
        \item Reconstruct the message digest using $R$.
        \item Complete the \wots chains from
          $\sigma_{\mathrm{\wots}}$ to recover the \wots public key.
        \item \ltree compression of the \wots public key gives us the
          $i$-th leaf of the \xmss tree.
        \item Use the authentication path to compute the root of the \xmss tree.
        \item Accept if the reconstructed root equals
          $\mathsf{root}$ in the public key.
      \end{enumerate}
    \end{column}
    \begin{column}{0.43\textwidth}
      \centering
      \vspace{2mm}
      \begin{tikzpicture}[x=0.68cm,y=0.7cm,font=\scriptsize]
        \foreach \i in {0,...,7}

\begin{frame}{\xmssmt: Multi-Tree Variant}
  \small
  \begin{itemize}
      \setlength{\itemsep}{2mm}
    \item \xmssmt replaces the single \xmss Merkle tree with a hypertree
      of $d$ layers; each tree has height $h/d$.
    \item \textbf{Key generation:} compute the root of the top tree
      in the hypertree to obtain the public root.
    \item \textbf{Signature generation:} include one \wots signature
      and its authentication path for each of the $d$ layers.
      At each upper layer $j=1,\ldots,d-1$, sign the root of the tree
      below; skip the message-digest computation.
    \item Allows more signatures while keeping running times manageable,
      at the expense of increased signature size.
  \end{itemize}
  % RFC 8391, Section 4.2. Upper-layer signatures can be reused for the
  % same lower-tree root; a \wots key must never sign distinct roots.
\end{frame}
