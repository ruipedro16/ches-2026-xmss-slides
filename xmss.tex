\begin{frame}{\xmss: Overview}
  \begin{itemize}
    \setlength{\itemsep}{4mm}
    \item XMSS~\cite{DBLP:conf/pqcrypto/BuchmannDH11} is a stateful hash-based signature scheme.
    \item Security relies on the security of the underlying hash function.
    \item XMSS combines a one-time signature scheme (\wots) with a Merkle tree.
    \vspace{2mm}
    \begin{itemize}
      \setlength{\itemsep}{1mm}
      \item The root of the Merkle tree is the public key of the XMSS instance.
      \item Each leaf corresponds to a (compressed) \wots public key.
      \item A Merkle tree of height $h$ has $2^h$ leaves, and thus allows signing $2^h$ messages.
    \end{itemize}
  \end{itemize}
\end{frame}

% \wots key generation and signing: RFC 8391, Sections 3.1.3--3.1.5.
\begin{frame}{\wots: Key Generation}
  \footnotesize
  \begin{enumerate}
    \setlength{\itemsep}{1mm}
    \item Generate $\ell$ secret $n$-byte values
      $\mathsf{sk}_0,\ldots,\mathsf{sk}_{\ell-1}$ (pseudorandomly)
    \item Each value starts a hash chain: apply the chaning function $w-1$ steps to obtain
      % $\mathsf{pk}_j=\mathsf{chain}(\mathsf{sk}_j,0,w-1)$.
      \item The chaining function is defined as $c_k^i(x,\mathbf{r}) = f_k(c_k^{i-1}(x,\mathbf{r})\oplus{r_i})$, with $c_k^0(x,\mathbf{r}) = x$, 
      where $\mathbf{r}$ are randomizers and $k$ is keying parameter for the hash function.\\
  \end{enumerate}
  \vspace{0mm}
  \begin{center}
    \centering
    \begin{tikzpicture}[x=1.05cm,y=0.65cm,font=\scriptsize,>=Latex]
      \foreach \j/\y in {0/1,1/0,\ell-1/-1} {
        \node[anchor=east,text=Gold] at (-0.3,\y) {$\mathsf{sk}_{\j}$};
        \foreach \k in {0,...,3}
          \node[xmss node] (k\y\k) at (\k,\y) {};
        \foreach \k/\next in {0/1,1/2,2/3}
          \draw[->,Teal] (k\y\k) -- (k\y\next);
        \node[anchor=west,text=Teal] at (3.3,\y) {$\mathsf{pk}_{\j}$};
      }
      \node at (1.5,-1.65) {Example: $w=4$};
    \end{tikzpicture}
  \end{center}
  \vspace{1mm}
  {\footnotesize
    \textbf{Secret key:} $\mathsf{SK}=(\mathsf{sk}_0,\ldots,\mathsf{sk}_{\ell-1})$\\[1mm]
    \textbf{Public key:} $\mathsf{PK}=(\mathsf{pk}_0,\ldots,\mathsf{pk}_{\ell-1})$\\[1mm]
    $n$: element size in bytes; $w$: Winternitz parameter;
    $\ell$: number of message and checksum digits.\\[1mm]
  }
\end{frame}

\begin{frame}{\wots: Signature Generation \& Verification}
  \small
  \begin{enumerate}
    \setlength{\itemsep}{2mm}
    \item Encode the message as base-$w$ digits
      $m_0,\ldots,m_{\ell_1-1}$
    \item Append the base-$w$ digits of the checksum
      $C=\sum_{j=0}^{\ell_1-1}(w-1-m_j)$, giving
      $\ell$ digits $b_0,\ldots,b_{\ell-1}\in\{0,\ldots,w-1\}$.
    \item The signature is $(\sigma_0,\ldots,\sigma_{\ell-1})$, where
      $\sigma_j=c_k^{b_j}(\mathsf{sk}_j,\mathbf{r})$.
    \item To verify a signature, it suffices to
      confirm that completing the chains permits reconstructing the \wots public key.
  \end{enumerate}
  \vspace{2mm}
  \begin{center}
    \begin{tikzpicture}[x=1.7cm,y=0.75cm,font=\scriptsize,>=Latex]
      \foreach \j/\y/\digit in {0/1/1,1/0/3,2/-1/2} {
        \node[anchor=east] at (-0.3,\y) {$b_{\j}=\digit$};
        \foreach \k in {0,...,3}
          \node[xmss node,draw=Muted,fill=white] (s\j\k) at (\k,\y) {};
        \foreach \k/\next in {0/1,1/2,2/3} {
          \ifnum\k<\digit
            \draw[->,Teal,thick] (s\j\k) -- (s\j\next);
          \else
            \draw[->,Muted,dashed] (s\j\k) -- (s\j\next);
          \fi
        }
        \node[xmss node,draw=Gold,fill=Gold!20] at (s\j\digit) {};
        \node[anchor=west,text=Gold] at (3.3,\y) {$\sigma_{\j}$};
      }
      \node at (1.5,-1.65) {Example: $w=4$; gold nodes form the signature.};
    \end{tikzpicture}
  \end{center}
  {\footnotesize
    \alert{Each \wots keypair signs at most one message.}
  }
\end{frame}

% L-tree compression: RFC 8391, Section 4.1.5, Algorithm 8.
\begin{frame}{\ltree: Compressing a \wots Public Key}
  \small
  \begin{columns}[T,onlytextwidth]
    \begin{column}{0.55\textwidth}
      \textbf{Input:} $\ell$ public-key elements, each $n$ bytes.
      \begin{enumerate}
        \setlength{\itemsep}{2.5mm}
        \item Hash adjacent pairs with randomized hashing (hash keys and masks are derived from
        \texttt{pub\_seed} and the \ltree address).
        \item If a level has an odd number of nodes, carry the
          last node to the next level unchanged.
        \item Repeat until a single $n$-byte node remains.
      \end{enumerate}
      
    \end{column}

  \end{columns}

\begin{frame}{\xmss: Key Generation}
  \small
  \setbeamerfont{itemize/enumerate body}{size=\footnotesize}
  \begin{columns}[T,onlytextwidth]
    \begin{column}{0.47\textwidth}
      \begin{enumerate}
        \setlength{\itemsep}{0.5mm}
        \setlength{\parsep}{0pt}
        \item Sample three random $n$-byte seeds:
          \begin{itemize}
            \footnotesize
            \setlength{\itemsep}{0.5mm}
            \setlength{\parsep}{0pt}
            \item \texttt{sk\_seed}: PRF seed for \wots secret keys;
              keeps the secret key compact.
            \item \texttt{sk\_prf}: PRF key for message-hashing randomizers.
            \item \texttt{pub\_seed}: PRF key for randomized-hashing bitmasks.
          \end{itemize}
        \item Initialize the signing-key index: $i=0$.
        \item Compute the tree root from the \wots public keys,
          (compressed into leaves via \ltree).
      \end{enumerate}
    \end{column}
    \begin{column}{0.50\textwidth}
      \centering
      \resizebox{\linewidth}{!}{\xmsskeygentree}\\[1mm]
      {\footnotesize $i=0,\ldots,2^h-1$}\\[1mm]
      \small Example: $h=3$ ($2^h$ leaves)

    \end{column}
  \end{columns}
  \vspace{0mm}
  {\footnotesize
    \textbf{Public key:} $\mathsf{PK}=(\mathsf{root},\mathtt{pub\_seed})$.\\
    \textbf{Secret key:} $\mathsf{SK}=(i,\mathtt{sk\_seed},\mathtt{sk\_prf},\mathsf{PK})$.\\
    \textbf{\xmssmt:} same process, but compute the root of the top tree
    in the hypertree; each tree has height $h/d$.
  }
\end{frame}

\begin{frame}{\xmss: Signature Generation}
  \small
  \begin{enumerate}
    \setlength{\itemsep}{2mm}
    \item Use the current secret-key index $i$ to select the one-time key;
      update the stored index to prevent key reuse.
    \item Derive the pseudorandom value $R$ using a PRF keyed by
      \texttt{sk\_prf}, and compute the message digest using $R$.
    \item Sign the digest with the selected \wots secret key.
    \item Compute the Merkle authentication path for leaf $i$:
      it authenticates the corresponding \wots public key.
  \end{enumerate}
  \begin{block}{Signature}
    $\sigma=(i,R,\sigma_{\mathrm{WOTS}^{+}},\mathsf{auth})$:
    index, randomizer, \wots signature, and authentication path.
  \end{block}
  {\footnotesize
    \textbf{\xmssmt:} include one \wots signature and authentication path
    per layer. For layers $j=1,\ldots,d-1$, sign the root of the tree
    below; the initial message digest is computed only once.
  }
\end{frame}

\begin{frame}{\xmss: Signature Verification}
  \relax
  {\small
    \textbf{Inputs}: message $M$, $\mathsf{PK}=(\mathsf{root},\mathsf{SEED})$,
    $\sigma=(i,R,\sigma_{\mathrm{\wots}},\mathsf{auth}_i)$, with $0\leq i<2^h$.\par}
  \vspace{2mm}
  \begin{columns}[T,onlytextwidth]
    \begin{column}{0.54\textwidth}
      \begin{enumerate}
        \small
        \setlength{\itemsep}{0.5mm}
        \item<1-> Recompute the message digest $D = H_{\mathrm{msg}}(R\mathbin\Vert\mathsf{root}\mathbin\Vert i,M)$.
        \item<2-> Complete the \wots chains from
          $\sigma_{\mathrm{\wots}}$ to recover $\mathsf{pk}_i$.
        \item<3->LTree compression gives leaf $i$.
        \item<4-> Hash upward with the $h$ authentication siblings.
          Bits of $i$ determine left/right order.
        \item<5-> Accept if the reconstructed root equals
          $\mathsf{root}$ in the public key.
      \end{enumerate}
    \end{column}
    \begin{column}{0.43\textwidth}
      \centering
      \vspace{2mm}
      \begin{tikzpicture}[x=0.68cm,y=0.7cm,font=\scriptsize]
        \foreach \i in {0,...,7}
          \node[xmss node,draw=Muted!45,fill=white] (v\i) at (\i,0) {};

\begin{frame}{\xmssmt: Multi-Tree Variant}
  \begin{columns}[T,onlytextwidth]
    \begin{column}{0.53\textwidth}
      \begin{itemize}
        \small
        \setlength{\itemsep}{2mm}
        \item \xmssmt generalizes \xmss by replacing the single Merkle tree with a hypertree of $d$
        layers, each of which contains a Merkle tree of height $\frac{h}{d}$\\
        \item Allows the total number of signatures to be increased while keeping running times
manageable, at the expense of increased signature size
      \end{itemize}
      \uncover<4->{%
        {\setbeamerfont{block title}{size=\small}
          \setbeamerfont{block body}{size=\footnotesize}
          \begin{block}{Tradeoff}
            $2^h$ messages; smaller trees to generate.\\[1mm]
            Signature: $d$ \wots signatures and $h$ path nodes.\\[1mm]
            For $d=1$, \xmssmt is functionally equivalent to \xmss.
          \end{block}}}
    \end{column}
    \begin{column}{0.44\textwidth}
      \centering
      \vspace{2mm}
      \begin{tikzpicture}[x=1cm,y=0.75cm,font=\scriptsize,>=Latex]
